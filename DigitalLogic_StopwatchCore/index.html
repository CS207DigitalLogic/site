    数字逻辑：综合项目（一）——秒表计时核心

    实验目标

    • 了解秒表项目的整体架构与模块划分
    • 实现 BCD 计数器（模10计数器，适配数码管显示）
    • 实现 4 级 BCD 计数器级联，组成 00.00 - 99.99 的计数链
    • 使用分频器产生 100Hz 基准时钟脉冲
    • 通过仿真和 LED 验证计时核心的正确性

    1. 秒表项目概述

    1.1 项目目标

    用 3 周时间完成一个完整的秒表（Stopwatch）系统，体验模块化开发流程：

    计时核心 → 显示与控制 → 系统集成

    1.2 秒表功能

    • 4 位数码管显示 SS.cc（秒.百分秒），范围 00.00 - 99.99
    • 两个按键：
    • Start/Stop：启动 / 暂停计时
    • Reset：暂停状态下清零
    • 精度：10ms（百分秒）

    1.3 模块架构

    本周任务：实现虚线框内上半部分——clk_div + BCD 计数器链。

    2. BCD 计数器

    2.1 为什么用BCD而不是二进制？

    秒表需要在数码管上显示十进制数字。如果用一个二进制计数器计到 9999，还需要做二进制到 BCD 的转换（比较复杂）。

    更简单的方案：直接用 4 个模10计数器（BCD计数器）级联，每个计数器输出一位十进制数（0-9），直接送入七段译码器。

    2.2 BCD 计数器代码

    这是 counter_mod，参数设为 10：

    module bcd_counter (
    input        clk,
    input        rst_n,
    input        en,
    output reg [3:0] cnt,
    output       carry
);
    assign carry = (cnt == 4'd9) && en;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            cnt <= 4'd0;
        else if (en) begin
            if (cnt == 4'd9)
                cnt <= 4'd0;
            else
                cnt <= cnt + 1'd1;
        end
    end
endmodule

    2.3 4级计数器级联

    tick_100hz → [百分秒个位] ─carry→ [百分秒十位] ─carry→ [秒个位] ─carry→ [秒十位]
               cnt0 (0-9)          cnt1 (0-9)         cnt2 (0-9)      cnt3 (0-9)

    显示格式：[cnt3][cnt2].[cnt1][cnt0] = SS.cc

    级联逻辑： 

    wire tick_100hz;
wire carry0, carry1, carry2;
wire [3:0] cnt0, cnt1, cnt2, cnt3;

// 100Hz 基准脉冲（10ms 一次）
clk_div #(.MAX_COUNT(1_000_000)) u_100hz (
    .clk(clk), .rst_n(rst_n), .tick(tick_100hz)
);

// 百分秒个位 (0-9)
bcd_counter u_cs0 (
    .clk(clk), .rst_n(rst_n), .en(tick_100hz),
    .cnt(cnt0), .carry(carry0)
);

// 百分秒十位 (0-9)
bcd_counter u_cs1 (
    .clk(clk), .rst_n(rst_n), .en(carry0),
    .cnt(cnt1), .carry(carry1)
);

// 秒个位 (0-9)
bcd_counter u_s0 (
    .clk(clk), .rst_n(rst_n), .en(carry1),
    .cnt(cnt2), .carry(carry2)
);

// 秒十位 (0-9)
bcd_counter u_s1 (
    .clk(clk), .rst_n(rst_n), .en(carry2),
    .cnt(cnt3), .carry()      // 最高位进位不使用
);

    2.4 进位时序分析

    以 09.99 → 10.00 为例（一个 tick_100hz 到来时）：

    信号 当前值 en 操作 新值
    cnt0 9 tick_100hz=1 9→0, carry0=1 0
    cnt1 9 carry0=1 9→0, carry1=1 0
    cnt2 9 carry1=1 9→0, carry2=1 0
    cnt3 0 carry2=1 0→1 1

    结果：10.00 ✓（所有更新在同一个时钟边沿完成）

    3. 分频器参数

    用途 目标频率 MAX_COUNT 说明
    百分秒基准 100Hz 1,000,000 100MHz / 1,000,000 = 100Hz
    仿真加速 — 10 仿真时使用

    步骤3：编写计时核心测试顶层

    创建一个临时的顶层模块，只包含分频器 + 计数器链 + LED输出，用于验证计时功能：

    module stopwatch_core_test (
    input        clk,
    input        rst_n,
    output [3:0] led_bg,     // 显示百分秒个位（cnt0）
    output [3:0] led_bs,     // 显示百分秒个位（cnt1）
    output [3:0] led_sg,     // 显示秒个位（cnt2）    
    output [3:0] led_ss     // 显示秒十位（cnt3）        
);
    wire tick_100hz;
    wire carry0, carry1, carry2;
    wire [3:0] cnt0, cnt1, cnt2, cnt3;

    clk_div #(.MAX_COUNT(1_000_000)) u_100hz (
        .clk(clk), .rst_n(rst_n), .tick(tick_100hz)
    );

    bcd_counter u_cs0 (.clk(clk), .rst_n(rst_n), .en(tick_100hz),
                        .cnt(cnt0), .carry(carry0));
    bcd_counter u_cs1 (.clk(clk), .rst_n(rst_n), .en(carry0),
                        .cnt(cnt1), .carry(carry1));
    bcd_counter u_s0  (.clk(clk), .rst_n(rst_n), .en(carry1),
                        .cnt(cnt2), .carry(carry2));
    bcd_counter u_s1  (.clk(clk), .rst_n(rst_n), .en(carry2),
                        .cnt(cnt3), .carry());

    assign led_bg = cnt0;
    assign led_bs = cnt1; 
    assign led_sg = cnt2;
    assign led_ss = cnt3;     
endmodule

    步骤4：仿真验证

    `timescale 1ns/1ps

module tb_stopwatch_core();
    reg clk, rst_n;
    wire [3:0] cnt0, cnt1, cnt2, cnt3;
    wire carry0, carry1, carry2;
    wire tick;

    clk_div #(.MAX_COUNT(10)) u_tick (
        .clk(clk), .rst_n(rst_n), .tick(tick)
    );

    bcd_counter u0 (.clk(clk), .rst_n(rst_n), .en(tick),
                     .cnt(cnt0), .carry(carry0));
    bcd_counter u1 (.clk(clk), .rst_n(rst_n), .en(carry0),
                     .cnt(cnt1), .carry(carry1));
    bcd_counter u2 (.clk(clk), .rst_n(rst_n), .en(carry1),
                     .cnt(cnt2), .carry(carry2));
    bcd_counter u3 (.clk(clk), .rst_n(rst_n), .en(carry2),
                     .cnt(cnt3), .carry());

    initial clk = 0;
    always #5 clk = ~clk;

    initial begin
        $monitor("time=%6d: %d%d.%d%d",
                 $time, cnt3, cnt2, cnt1, cnt0);
        rst_n = 0; #30; rst_n = 1;
        // 等待计到 01.00 (100个tick = 100*10*10ns = 10000ns)
        #120000;
        $finish;
    end
endmodule

    仿真观察要点： - 计数序列：00.00 → 00.01 → ... → 00.09 → 00.10 → ... → 00.99 → 01.00 - 进位是否正确：百分秒满100后秒加1
